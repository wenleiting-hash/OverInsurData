export interface AppointmentRecord {
  id: string                    // ap1, ap2... ap12
  channelId: string             // c1, c2... 
  channelName: string           // "Pacific Coast Insurance Group"
  channelNpn: string            // "NPN12348901"
  insurerId: string             // 1, 2...
  insurerName: string           // "Travelers", "Chubb"...
  insurerShort: string          // "Travelers", "Chubb"...
  state: string                 // CA, NY, TX (US 州缩写)
  line: string                  // P&C, Auto, Commercial...
  status: AppointmentStatus     // 见下文枚举
  submittedDate: string         // ISO date format (YYYY-MM-DD)
  approvedDate?: string
  expiryDate: string
  terminatedDate?: string
  terminationReason?: string
  renewalStatus?: RenewalStatus
  daysToExpiry: number          // 距离过期天数（负数=已过期）
  submittedBy: string           // "Sarah Chen"

  rejectionReason?: string      // 被拒原因
  niprTransactionId?: string    // "NIPR-2023-04581"

export type AppointmentStatus = 
  | 'approved'       // 已批准
  | 'pending'        // 待审核
  | 'under-review'   // 审核中
  | 'rejected'       // 已拒绝
  | 'expired'        // 已过期
  | 'terminated';    // 已终止

export type RenewalStatus = 
  | 'not-due'        // 未到期
  | 'due-soon'       // 即将到期（60 天内）
  | 'in-progress'    // 续期中
  | 'renewed';       // 已续期

### **数据统计辅助函数**

```typescript
// 统计 Appointment 状态分布
export const countAppointmentStatus = (apps: AppointmentRecord[]): Record<string, number> => ({
  total: apps.length,
  approved: apps.filter(a => a.status === 'approved').length,
  pending: apps.filter(a => a.status === 'pending').length,
  underReview: apps.filter(a => a.status === 'under-review').length,
  rejected: apps.filter(a => a.status === 'rejected').length,
  expired: apps.filter(a => a.status === 'expired').length,
  terminated: apps.filter(a => a.status === 'terminated').length
})

// 计算平均处理天数
export const calculateAvgProcessingDays = (apps: AppointmentRecord[]): number => {
  const validApps = apps.filter(a => a.processingDays !== undefined);
  return Math.round(
    validApps.reduce((sum, a) => sum + (a.processingDays || 0), 0) / 
    Math.max(1, validApps.length)
  );
};

// 查找即将到期的 Appointment（60 天内）
export const findExpiringSoon = (apps: AppointmentRecord[], threshold: number = 60): AppointmentRecord[] => {
  return apps.filter(a => a.daysToExpiry > 0 && a.daysToExpiry <= threshold && a.renewalStatus === 'due-soon');
};
```
