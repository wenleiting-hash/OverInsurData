## 六、关键配置文件

### 6.1 docker-compose.yml

```yaml
# /opt/overinsur/docker-compose.yml
version: "3.8"

services:
  overinsur-pg:
    image: postgres:16
    container_name: overinsur-pg
    restart: unless-stopped
    environment:
      POSTGRES_DB: overinsur_db
      POSTGRES_USER: overinsur
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    ports:
      - "127.0.0.1:5460:5432"
    volumes:
      - overinsur-pg-data:/var/lib/postgresql/data
      - /opt/overinsur/backup:/backup
    mem_limit: 512m
    networks:
      - overinsur-net

  carrier-api:
    image: overinsur/carrier-api:1.0.0
    container_name: carrier-api
    restart: unless-stopped
    depends_on:
      - overinsur-pg
    environment:
      NODE_ENV: production
      DB_HOST: overinsur-pg
      DB_PORT: 5432
      DB_NAME: overinsur_db
      DB_USER: overinsur
      DB_PASSWORD: ${DB_PASSWORD}
      JWT_SECRET: ${JWT_SECRET}
      JWT_REFRESH_SECRET: ${JWT_REFRESH_SECRET}
      INTEGRATION_MASTER_KEY: ${INTEGRATION_MASTER_KEY}
      CORS_ORIGINS: http://220.154.128.28:8601
      WORKOS_MOCK: "1"
      UPLOAD_DIR: /app/uploads
    ports:
      - "127.0.0.1:8610:8080"
    volumes:
      - /opt/overinsur/data/uploads:/app/uploads
    mem_limit: 512m
    networks:
      - overinsur-net

networks:
  overinsur-net:
    driver: bridge

volumes:
  overinsur-pg-data:
```

### 6.2 Nginx 配置

```nginx
# /etc/nginx/conf.d/overinsur.conf
server {
    listen 8601;
    server_name _;
    client_max_body_size 25m;
    root /opt/overinsur/deploy/web;
    index index.html;

    # SPA history fallback
    location / {
        try_files $uri $uri/ /index.html;
    }

    # 后端 API
    location /api/ {
        proxy_pass http://127.0.0.1:8610;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 上传文件
    location /uploads/ {
        proxy_pass http://127.0.0.1:8610;
        proxy_set_header Host $host;
    }

    # gzip
    gzip on;
    gzip_types text/css application/javascript application/json;
    gzip_min_length 1024;
}
```

## 九、风险与应对

| 风险 | 影响 | 应对 |
|---|---|---|
| Swap 100% 占满 | 新容器可能 OOM | 设 mem_limit 512M；新增约 1G 在 7.1G 可用量内 |
| 前端 8 个 TS 错误阻断构建 | 无法产出 dist | 阶段 1 优先修复 |
| DB 命名不一致（ai_saas vs overinsur_db） | 连接失败 | 统一用 overinsur_db |
| 迁移文件为空 | 无法自动迁移 | 手写完整建表 DDL |
| 多租户共享服务器 | 安全风险 | 所有服务只绑 127.0.0.1，仅 Nginx 暴露 |
