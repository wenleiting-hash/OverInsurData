### 1.4 公共前置步骤

所有 API 测试均需先获取 JWT Token：

```powershell
$loginResp = Invoke-RestMethod -Uri "http://localhost:3001/api/auth/login" `
  -Method POST -ContentType "application/json" `
  -Body '{"username":"admin","password":"admin123"}'
$token = $loginResp.data.accessToken
$headers = @{ Authorization = "Bearer $token" }
```

### A. 测试数据检查 SQL

```sql
-- 验证保险公司数据
SELECT COUNT(*) AS total, 
  COUNT(CASE WHEN status = 'active' THEN 1 END) AS active_count,
  COUNT(CASE WHEN status = 'inactive' THEN 1 END) AS inactive_count
FROM insurance_carrier WHERE deleted = FALSE;

-- 验证产品数据
SELECT p.line_of_business, COUNT(*) AS product_count,
  SUM(p.premium_ytd) AS total_premium
FROM insurance_product p
WHERE p.deleted = FALSE
GROUP BY p.line_of_business
ORDER BY total_premium DESC;

-- 验证合作关系数据
SELECT cp.cooperation_type, cp.status, COUNT(*) AS coop_count
FROM carrier_partnership cp
WHERE cp.deleted = FALSE
GROUP BY cp.cooperation_type, cp.status;

-- 验证合同数据
SELECT cc.contract_type, cc.status, COUNT(*) AS contract_count
FROM carrier_contract cc
WHERE cc.deleted = FALSE
GROUP BY cc.contract_type, cc.status;

-- 验证对接人数据
SELECT ccc.role, COUNT(*) AS contact_count
FROM carrier_contact ccc
WHERE ccc.deleted = FALSE
GROUP BY ccc.role;

-- 验证结算配置
SELECT csc.cycle, COUNT(*) AS config_count
FROM carrier_settlement_config csc
WHERE csc.deleted = FALSE
GROUP BY csc.cycle;
```
