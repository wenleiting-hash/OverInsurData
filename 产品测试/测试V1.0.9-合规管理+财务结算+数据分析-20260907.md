$headers = @{ Authorization = "Bearer $token" }
# 后端直接访问: http://localhost:8080/api/...

验证 SQL：
```sql
-- 确认 10 张表已创建
SELECT tablename FROM pg_tables
WHERE schemaname = 'public'
  AND tablename IN (
    'compliance_rule','ofac_screening_record','compliance_interception',
    'nipr_license','license_expiry_reminder','compliance_audit_report',
    'commission_bill','commission_bill_line','reconciliation_diff','settlement_cycle_config'
  )
ORDER BY tablename;

-- 确认合规规则种子数据
SELECT rule_id, rule_name, category, action, enabled, priority
FROM compliance_rule WHERE deleted = FALSE ORDER BY priority;
```
