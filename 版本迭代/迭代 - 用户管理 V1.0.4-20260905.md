// 1. Role-based Access Control (RBAC)
@Injectable()
export class PermissionGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const user = this.extractUser(context);
    const requiredPermissions = this.getRequiredPermissions(context);
    
    return user.permissions.some(perm => 
      requiredPermissions.includes(perm)
    );
  }
}

## ⚠️ 数据迁移策略

### Mock Data → Real Database Mapping

| Mock Field | DB Column | Notes |
|------------|-----------|-------|
| u1/u2... | user_uuid | Generate UUID via gen_random_uuid() |
| admin/li.xiaoyan | username | UNIQUE constraint |
| P@ssw0rd2026! | password_hash | BCrypt hash (pre-computed in SQL) |
| tech/ops... | dept_code | JOIN with auth_department table |
| local/sso/ldap | auth_method | CHECK constraint validation |
| active/inactive... | status | Status enum validation |
| roles array | auth_user_role | Many-to-many mapping |

### Migration Script Execution Order

```bash
# Step 1: Create database schema
psql -U admin -d db_auth -f V1.0.0__create_auth_tables.sql

# Step 2: Seed departments & roles
psql -U admin -d db_auth -f R__seed_auth_data.sql

# Step 3: Migrate mock users
psql -U admin -d db_auth -f R__seed_mock_users.sql

# Step 4: Verify data integrity
psql -U admin -d db_auth <<EOF
SELECT COUNT(*) FROM auth_user WHERE deleted = FALSE;
SELECT username, dept_code, STRING_AGG(role_key, ',') as roles
FROM auth_user au
JOIN auth_user_role aur ON au.id = aur.user_id
JOIN auth_role ar ON aur.role_id = ar.role_id
GROUP BY username, dept_code
ORDER BY au.id;
EOF
```
